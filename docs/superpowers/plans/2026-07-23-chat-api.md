# Chat API (viper7) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship the backend for a streaming, tool-using RAG chat assistant
(viper7) grounded in the course knowledge base.

**Architecture:** `POST /api/chat` runs `streamText` on Gemini 3.6 Flash via the
Vercel AI Gateway with two tools (`searchKB` retrieval + enrichment,
`checkFlyability` weather), a DB-persona system prompt, and persists turns to
`aiChats`/`aiMessages`. `GET /api/chats` and `GET /api/chats/:id` read history.

**Tech Stack:** TanStack Start server handlers, AI SDK v6 (`ai@6`), Vercel AI
Gateway (bare model-id strings), `@ai-sdk/google` embeddings, Drizzle + Postgres
+ pgvector, better-auth, zod, vitest.

Design spec: `docs/superpowers/specs/2026-07-23-chat-api-design.md`.

- [ ] **Step 1: Implement the query** (`src/db/knowledge-base.ts`, double quotes, `@/db`):

```ts
import { embed } from "ai";
import { and, cosineDistance, desc, eq, gt, isNull, or, sql } from "drizzle-orm";
import { db } from "@/db";
import { docs } from "@/db/schema";
import { embeddingModel } from "#/ai/gemini";

export type KBResult = { chunk: string; heading: string | null; similarity: number };

/**
 * Cosine-similarity retrieval over ingested `docs` embeddings for a query.
 * Scopes to a course plus org-wide (null courseId) docs. `1 - cosineDistance`
 * = cosine similarity; filtered by `minScore` and ordered desc.
 */
export async function searchKB(
  query: string,
  opts: { maxResults?: number; minScore?: number; courseId?: number } = {},
): Promise<KBResult[]> {
  const { maxResults = 5, minScore = 0, courseId } = opts;
  const { embedding } = await embed({ model: embeddingModel, value: query });
  const similarity = sql<number>`1 - (${cosineDistance(docs.embedding, embedding)})`;
  const courseScope =
    courseId == null
      ? undefined
      : or(eq(docs.courseId, courseId), isNull(docs.courseId));
  return db
    .select({ chunk: docs.chunk, heading: docs.heading, similarity })
    .from(docs)
    .where(and(courseScope, gt(similarity, minScore)))
    .orderBy((t) => desc(t.similarity))
    .limit(maxResults);
}
```

```ts
type LessonContent = { name: string; text: string | null; proTips: string | null };
type ModuleContent = { name: string; lessons: LessonContent[] };
type CourseContent = { name: string; modules: ModuleContent[] };

export function courseContentToHtml(course: CourseContent): string {
  const esc = (s: string) => s.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;");
  const lesson = (l: LessonContent) =>
    `<h3>${esc(l.name)}</h3>${l.text ? `<p>${esc(l.text)}</p>` : ""}${l.proTips ? `<p><strong>Pro tips:</strong> ${esc(l.proTips)}</p>` : ""}`;
  const mod = (m: ModuleContent) => `<h2>${esc(m.name)}</h2>${m.lessons.map(lesson).join("\n")}`;
  return `<h1>${esc(course.name)}</h1>${course.modules.map(mod).join("\n")}`;
}
```

  topics as `<h2>title</h2>content`) and `makeSearchKBTool` whose `execute`:

  writes a transient `data-notification` "Thinking…" via `writer`, runs the
  three sources with `Promise.all`, returns `buildKBContext(...)`.

```ts
import { tool } from 'ai';
import { z } from 'zod';
import { getCourseContentForAgent } from '#/db/course-content';
import { getAllHelpTopics } from '#/db/help-topics';
import { searchKB, type KBResult } from '#/db/knowledge-base';

export const SearchKBParamsSchema = z.object({
  query: z.string().describe("the user's question"),
  maxResults: z.number().optional(),
  minScore: z.number().optional(),
});

export function buildKBContext(input: {
  kbResults: KBResult[];
  courseHtml: string;
  helpTopics: { title: string; content: string }[];
}): string {
  const kb = input.kbResults.map((r) => r.chunk).join('\n\n');
  const help = input.helpTopics.map((h) => `<h2>${h.title}</h2>${h.content}`).join('\n\n');
  return [input.courseHtml, kb, `<h1>Help</h1>\n${help}`].filter(Boolean).join('\n\n');
}

export function makeSearchKBTool(opts: {
  writer?: { write: (p: unknown) => void };
  courseSlug?: string;
  courseId?: number;
} = {}) {
  const courseSlug = opts.courseSlug ?? '3d-airmanship';
  return tool({
    description:
      'Search the comprehensive knowledge base about the course, drones, aircraft & airmanship. Always call this FIRST for any course/airmanship question, then supplement with general aviation knowledge if needed.',
    inputSchema: SearchKBParamsSchema,
    execute: async ({ query, maxResults, minScore }) => {
      if (query.trim().length < 3) {
        return "The user's query is too short to search the knowledge base. Decide how to respond.";
      }
      opts.writer?.write({ type: 'data-notification', data: { text: 'Thinking...' }, transient: true });
      const [kbResults, courseHtml, helpTopics] = await Promise.all([
        searchKB(query, { maxResults: maxResults ?? 5, minScore: minScore ?? 0, courseId: opts.courseId }),
        getCourseContentForAgent(courseSlug),
        getAllHelpTopics(),
      ]);
      return buildKBContext({ kbResults, courseHtml, helpTopics });
    },
  });
}
```
